# CRM运营平台 - 前端代码规范

## 1. 概述

本文档定义了CRM运营平台前端的代码编写规范，所有开发人员必须严格遵守，以确保代码质量、可读性和可维护性。

### 1.1 产品层级

```
平台 (Platform)
  └── 应用 (Application)
      └── 功能 (Feature)
          └── 操作 (Operation)
```

**当前平台：** CRM运营平台（CRM Operation Platform）

**架构原则：** 一个应用对应一个微服务 + 一个数据库 + 一个微前端

## 2. 基础规范

### 2.1 总体原则

1. **遵循Vue官方风格指南**：作为基础规范
2. **保持代码简洁**：单个函数不超过50行，单个文件不超过500行
3. **组件化开发**：合理拆分组件，提高复用性
4. **响应式优先**：充分利用Vue 3 Composition API
5. **类型安全**：推荐使用TypeScript

### 2.2 开发工具配置

#### IDE设置
- **编辑器**：VSCode（推荐）
- **编码格式**：UTF-8
- **缩进**：2个空格（不使用Tab）
- **行宽限制**：120字符
- **自动格式化**：保存时自动格式化代码

#### 必备插件
- **Volar**：Vue 3官方VSCode扩展
- **ESLint**：代码质量检查
- **Prettier**：代码格式化
- **Vue DevTools**：Vue调试工具

#### script setup推荐写法

```vue
<script setup>
import { ref, reactive, computed, onMounted } from 'vue';

  },

    const res = await getTenantList();

  fetchData();

  fetchData
```

#### 组合式函数（Composables）

```javascript
// composables/use-tenant.js
import { ref, computed } from 'vue';
import { getTenantList, getTenantById } from '@/api/tenant';

export function useTenant() {
  const loading = ref(false);
  const tenantList = ref([]);
  const currentTenant = ref(null);
  
  // 获取租户列表
  const fetchTenantList = async (params) => {
    loading.value = true;
    try {
      const res = await getTenantList(params);
      tenantList.value = res.data;
      return res;
    } catch (error) {
      console.error('获取租户列表失败', error);
      throw error;
    } finally {
      loading.value = false;
    }
  };
  
  // 获取租户详情
  const fetchTenantById = async (id) => {
    loading.value = true;
    try {
      const res = await getTenantById(id);
      currentTenant.value = res.data;
      return res;
    } catch (error) {
      console.error('获取租户详情失败', error);
      throw error;
    } finally {
      loading.value = false;
    }
  };
  
  // 活跃租户列表
  const activeTenants = computed(() => {
    return tenantList.value.filter(item => item.status === 1);
  });
  
  return {
    loading,
    tenantList,
    currentTenant,
    activeTenants,
    fetchTenantList,
    fetchTenantById
  };
}
```

```vue
<template>
  <div
```

## 6. API接口规范

### 6.1 Axios实例配置

```javascript
// api/request.js
import axios from 'axios';
import { ElMessage } from 'element-plus';
import router from '@/router';

// 创建axios实例
const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

// 请求拦截器
request.interceptors.request.use(
  config => {
    // 添加Token
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  error => {
    return Promise.reject(error);
  }
);

// 响应拦截器
request.interceptors.response.use(
  response => {
    const { code, message, data } = response.data;
    
    if (code === 200) {
      return data;
    } else {
      ElMessage.error(message || '请求失败');
      return Promise.reject(new Error(message));
    }
  },
  error => {
    if (error.response?.status === 401) {
      // 未授权，跳转登录
      ElMessage.error('登录已过期，请重新登录');
      router.push('/login');
    } else {
      ElMessage.error(error.message || '网络错误');
    }
    return Promise.reject(error);
  }
);

export default request;
```

```javascript
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import { userApi } from '@/api';

export const useUserStore = defineStore('user', () => {
  // State
  const userInfo = ref(null);
  const token = ref(localStorage.getItem('token') || '');
  const permissions = ref([]);
  
  // Getters
  const isLoggedIn = computed(() => !!token.value);
  const userName = computed(() => userInfo.value?.name || '');
  
  // Actions
  const login = async (credentials) => {
    try {
      const data = await userApi.login(credentials);
      token.value = data.token;
      userInfo.value = data.user;
      permissions.value = data.permissions;
      
      // 存储到localStorage
      localStorage.setItem('token', data.token);
      
      return data;
    } catch (error) {
      throw error;
    }
  };
  
  const logout = () => {
    token.value = '';
    userInfo.value = null;
    permissions.value = [];
    localStorage.removeItem('token');
  };
  
  const hasPermission = (permission) => {
    return permissions.value.includes(permission);
  };
  
  return {
    // State
    userInfo,
    token,
    permissions,
    // Getters
    isLoggedIn,
    userName,
    // Actions
    login,
    logout,
    hasPermission
  };
});
```

### 8.2 路由守卫

```javascript
// router/index.js
import { useUserStore } from '@/stores/user-store';

router.beforeEach((to, from, next) => {
  const userStore = useUserStore();
  
  // 设置页面标题
  document.title = to.meta.title || 'CRM运营平台';
  
  // 需要认证的路由
  if (to.meta.requiresAuth) {
    if (!userStore.isLoggedIn) {
      // 未登录，跳转登录页
      next({
        path: '/login',
        query: { redirect: to.fullPath }
      });
      return;
    }
    
    // 检查权限
    if (to.meta.permission && !userStore.hasPermission(to.meta.permission)) {
      // 无权限，跳转403页面
      next('/403');
      return;
    }
  }
  
  next();
});
```
