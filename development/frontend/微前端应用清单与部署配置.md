# CRM运营平台 - 微前端应用清单与部署配置

## 1. 概述

本文档详细说明了CRM运营平台所有微前端应用的清单、技术栈、端口分配和部署配置。

### 1.1 架构原则

**一个应用对应一个微服务 + 一个数据库 + 一个微前端**

```
前端微应用 ↔ 后端微服务 ↔ 独立数据库
```

### 1.2 技术栈统一

所有微前端应用采用统一的技术栈：

- **Vue 3.3+**：前端框架
- **Vite 4.4+**：构建工具
- **Element Plus 2.3+**：UI组件库
- **Pinia 2.1+**：状态管理
- **Vue Router 4.2+**：路由管理

```javascript
const microApps = [

    }

  {
    name: 'application-management',
    entry: getEntry('application-management'),
    container: '#micro-app-container',
    activeRule: '/application',
    props: {
      routerBase: '/application'
    }
  },
  {
    name: 'role-template-management',
    entry: getEntry('role-template-management'),
    container: '#micro-app-container',
    activeRule: '/role-template',
    props: {
      routerBase: '/role-template'
    }
  }
];

function getEntry(appName) {
  if (import.meta.env.MODE === 'development') {
    const portMap = {
      'operation-system': 5174,
      'tenant-management': 5175,
      'application-management': 5176,
      'role-template-management': 5177
    };
    return `//localhost:${portMap[appName]}`;
  }
  
  // 生产环境
  return `${window.location.origin}/${appName.replace('-management', '')}`;
}

export default microApps;
```

### 5.3 main.js（qiankun生命周期）

```javascript
import './public-path';
import { createApp } from 'vue';
import { createRouter, createWebHistory } from 'vue-router';
import { createPinia } from 'pinia';
import ElementPlus from 'element-plus';
import 'element-plus/dist/index.css';
import App from './App.vue';
import routes from './router';

let app = null;
let router = null;
let history = null;

function render(props = {}) {
  const { container } = props;
  
  // 创建路由实例
  history = createWebHistory(
    window.__POWERED_BY_QIANKUN__ ? '/tenant' : '/'
  );
  router = createRouter({
    history,
    routes
  });
  
  // 创建Vue实例
  app = createApp(App);
  app.use(router);
  app.use(createPinia());
  app.use(ElementPlus);
  
  // 挂载应用
  const containerElement = container 
    ? container.querySelector('#app') 
    : document.getElementById('app');
  app.mount(containerElement);
}

// 独立运行
if (!window.__POWERED_BY_QIANKUN__) {
  render();
}

// qiankun生命周期钩子
export async function bootstrap() {
  console.log('[tenant-management] app bootstraped');
}

export async function mount(props) {
  console.log('[tenant-management] props from main', props);
  render(props);
}

export async function unmount() {
  app?.unmount();
  app = null;
  router = null;
  history = null;
}
```

## 7. 构建配置

### 7.1 构建脚本

**scripts/build-all.js**

```javascript
import { spawn } from 'child_process';
import ora from 'ora';
import chalk from 'chalk';

const apps = [
  { name: '主应用', dir: 'main-app' },
  { name: '系统管理', dir: 'micro-apps/system-management' },
  { name: '租户管理', dir: 'micro-apps/tenant-management' },
  { name: '应用管理', dir: 'micro-apps/application-management' },
  { name: '角色权限模板', dir: 'micro-apps/role-template-management' }
];

async function buildApp(app) {
  const spinner = ora(`构建${app.name}...`).start();
  
  return new Promise((resolve, reject) => {
    const build = spawn('pnpm', ['build'], {
      cwd: app.dir,
      shell: true
    });
    
    build.on('close', (code) => {
      if (code === 0) {
        spinner.succeed(chalk.green(`${app.name}构建成功`));
        resolve();
      } else {
        spinner.fail(chalk.red(`${app.name}构建失败`));
        reject(new Error(`${app.name}构建失败`));
      }
    });
  });
}

async function buildAll() {
  console.log(chalk.blue('开始构建所有应用...\n'));
  
  for (const app of apps) {
    try {
      await buildApp(app);
    } catch (error) {
      console.error(error);
      process.exit(1);
    }
  }
  
  console.log(chalk.green('\n所有应用构建完成！'));
}

buildAll();
```

### 7.2 并行构建脚本

**scripts/build-parallel.js**

```javascript
import { spawn } from 'child_process';
import chalk from 'chalk';

const apps = [
  { name: '主应用', dir: 'main-app' },
  { name: '系统管理', dir: 'micro-apps/system-management' },
  { name: '租户管理', dir: 'micro-apps/tenant-management' },
  { name: '应用管理', dir: 'micro-apps/application-management' },
  { name: '角色权限模板', dir: 'micro-apps/role-template-management' }
];

async function buildParallel() {
  console.log(chalk.blue('开始并行构建所有应用...\n'));
  
  const promises = apps.map(app => {
    return new Promise((resolve, reject) => {
      console.log(chalk.yellow(`开始构建${app.name}...`));
      
      const build = spawn('pnpm', ['build'], {
        cwd: app.dir,
        shell: true
      });
      
      build.stdout.on('data', (data) => {
        console.log(`[${app.name}] ${data}`);
      });
      
      build.stderr.on('data', (data) => {
        console.error(`[${app.name}] ${data}`);
      });
      
      build.on('close', (code) => {
        if (code === 0) {
          console.log(chalk.green(`${app.name}构建成功\n`));
          resolve();
        } else {
          console.error(chalk.red(`${app.name}构建失败\n`));
          reject(new Error(`${app.name}构建失败`));
        }
      });
    });
  });
  
  try {
    await Promise.all(promises);
    console.log(chalk.green('所有应用构建完成！'));
  } catch (error) {
    console.error(chalk.red('部分应用构建失败'));
    process.exit(1);
  }
}

buildParallel();
```

## 8. 部署配置

### 8.1 Docker配置

**Dockerfile**

```dockerfile
# 构建阶段
FROM node:16-alpine as builder

WORKDIR /app

# 安装pnpm
RUN npm install -g pnpm@8

# 复制依赖文件
COPY package.json pnpm-lock.yaml pnpm-workspace.yaml ./
COPY main-app/package.json ./main-app/
COPY micro-apps/operation-system/package.json ./micro-apps/operation-system/
COPY micro-apps/tenant-management/package.json ./micro-apps/tenant-management/
COPY micro-apps/application-management/package.json ./micro-apps/application-management/
COPY micro-apps/role-template-management/package.json ./micro-apps/role-template-management/
COPY shared/package.json ./shared/

# 安装依赖
RUN pnpm install --frozen-lockfile

# 复制源代码
COPY . .

# 构建所有应用
RUN pnpm build:all

# 运行阶段
FROM nginx:alpine

# 复制构建产物
COPY --from=builder /app/main-app/dist /usr/share/nginx/html/
COPY --from=builder /app/micro-apps/operation-system/dist /usr/share/nginx/html/system
COPY --from=builder /app/micro-apps/tenant-management/dist /usr/share/nginx/html/tenant
COPY --from=builder /app/micro-apps/application-management/dist /usr/share/nginx/html/application
COPY --from=builder /app/micro-apps/role-template-management/dist /usr/share/nginx/html/role-template

# 复制Nginx配置
COPY nginx.conf /etc/nginx/nginx.conf

EXPOSE 80

CMD ["nginx", "-g", "daemon off;"]
```

### 8.2 Nginx配置

**nginx.conf**

```nginx
user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;
    
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';
    
    access_log /var/log/nginx/access.log main;
    
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;
    
    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml text/javascript 
               application/json application/javascript application/xml+rss 
               application/rss+xml application/atom+xml image/svg+xml 
               text/x-component text/x-js;
    
    server {
        listen 80;
        server_name operation.example.com;
        
        # 主应用
        location / {
            root /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
            
            # 禁用缓存（HTML文件）
            add_header Cache-Control "no-cache, no-store, must-revalidate";
            add_header Pragma "no-cache";
            add_header Expires "0";
        }
        
        # 系统管理微应用
        location /system {
            alias /usr/share/nginx/html/system;
            try_files $uri $uri/ /system/index.html;
            add_header Cache-Control "no-cache, no-store, must-revalidate";
        }
        
        # 租户管理微应用
        location /tenant {
            alias /usr/share/nginx/html/tenant;
            try_files $uri $uri/ /tenant/index.html;
            add_header Cache-Control "no-cache, no-store, must-revalidate";
        }
        
        # 应用管理微应用
        location /application {
            alias /usr/share/nginx/html/application;
            try_files $uri $uri/ /application/index.html;
            add_header Cache-Control "no-cache, no-store, must-revalidate";
        }
        
    # 角色权限模板微应用
    location /role-template {
        alias /usr/share/nginx/html/role-template;
        try_files $uri $uri/ /role-template/index.html;
        add_header Cache-Control "no-cache, no-store, must-revalidate";
    }
        
        # 静态资源缓存（JS、CSS、图片等）
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
            root /usr/share/nginx/html;
            expires 30d;
            add_header Cache-Control "public, immutable";
            access_log off;
        }
        
        # API代理
        location /api/ {
            proxy_pass http://gateway:18100/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # WebSocket支持
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }
    }
}
```

## 9. CI/CD配置

### 9.1 Jenkins Pipeline

**Jenkinsfile**

```groovy
pipeline {
    agent any
    
    environment {
        DOCKER_REGISTRY = 'registry.example.com'
        IMAGE_NAME = 'crm-operation-web'
        K8S_NAMESPACE = 'crm-operation'
    }
    
    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', 
                    url: 'https://github.com/your-org/crm-operation-web.git',
                    credentialsId: 'github-credentials'
            }
        }
        
        stage('Install Dependencies') {
            steps {
                sh 'npm install -g pnpm@8'
                sh 'pnpm install --frozen-lockfile'
            }
        }
        
        stage('Lint') {
            steps {
                sh 'pnpm lint'
            }
        }
        
        stage('Test') {
            steps {
                sh 'pnpm test'
            }
        }
        
        stage('Build') {
            steps {
                sh 'pnpm build:all'
            }
        }
        
        stage('Build Docker Image') {
            steps {
                script {
                    def imageTag = "${env.BUILD_NUMBER}"
                    sh "docker build -t ${DOCKER_REGISTRY}/${IMAGE_NAME}:${imageTag} ."
                    sh "docker tag ${DOCKER_REGISTRY}/${IMAGE_NAME}:${imageTag} ${DOCKER_REGISTRY}/${IMAGE_NAME}:latest"
                }
            }
        }
        
        stage('Push Docker Image') {
            steps {
                script {
                    def imageTag = "${env.BUILD_NUMBER}"
                    sh "docker push ${DOCKER_REGISTRY}/${IMAGE_NAME}:${imageTag}"
                    sh "docker push ${DOCKER_REGISTRY}/${IMAGE_NAME}:latest"
                }
            }
        }
        
        stage('Deploy to K8S') {
            steps {
                script {
                    def imageTag = "${env.BUILD_NUMBER}"
                    sh """
                        kubectl set image deployment/crm-operation-web \
                        web=${DOCKER_REGISTRY}/${IMAGE_NAME}:${imageTag} \
                        -n ${K8S_NAMESPACE}
                    """
                    sh "kubectl rollout status deployment/crm-operation-web -n ${K8S_NAMESPACE}"
                }
            }
        }
    }
    
    post {
        success {
            echo '部署成功！'
        }
        failure {
            echo '部署失败！'
        }
    }
}
```
