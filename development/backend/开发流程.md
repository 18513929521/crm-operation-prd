### 1.1 技术要求

**开发环境：**
- **Java**: 17+ (LTS版本)
- **Maven**: 3.8+
- **Python**: 3.12+ (用于脚本和集成测试)
- **Git**: 2.30+
- **Docker**: 24.x+

### 1.2 适用范围

- 所有微服务应用的开发
- 功能特性的新增和修改
- Bug修复和性能优化

- [系统架构文档](./系统架构文档.md)

**scripts/deploy/deploy_nacos_config.py**
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Nacos配置部署脚本
Requirements: Python 3.12+
"""
import sys
import requests
import yaml
from pathlib import Path
from typing import Optional

# 检查Python版本
if sys.version_info < (3, 12):
    raise RuntimeError("Python 3.12+ is required")

def upload_config(
    server_addr: str,
    namespace: str,
    group: str,
    data_id: str,
    content: str,
    username: str = "nacos",
    password: str = "nacos"
) -> bool:
    """
    上传配置到Nacos
    
    Args:
        server_addr: Nacos服务器地址
        namespace: 命名空间ID
        group: 分组名称
        data_id: 配置文件ID
        content: 配置内容
        username: Nacos用户名
        password: Nacos密码
        
    Returns:
        bool: 上传是否成功
    """
    url = f"{server_addr}/nacos/v1/cs/configs"
    data = {
        "dataId": data_id,
        "group": group,
        "tenant": namespace,
        "content": content,
        "type": "yaml"
    }
    
    # 添加认证
    auth = (username, password)
    
    try:
        response = requests.post(url, data=data, auth=auth, timeout=10)
        return response.text == "true"
    except Exception as e:
        print(f"Error uploading config: {e}")
        return False

def deploy_configs(env: str = "dev"):
    """部署配置文件到指定环境"""
    server_addr = "http://localhost:8848"
    namespace = f"operation-{env}"
    config_dir = Path(f"config/nacos/operation-{env}")
    
    for config_file in config_dir.glob("*.yaml"):
        with open(config_file, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 确定Group
        filename = config_file.stem
        if filename.startswith("common-"):
            group = "COMMON_GROUP"
        else:
            # 从文件名推导Group，如 crm-operation-tenant.yaml -> TENANT_GROUP
            app_name = filename.replace("crm-operation-", "").upper()
            group = f"{app_name}_GROUP"
        
        data_id = config_file.name
        
        print(f"Uploading {data_id} to {namespace}/{group}...")
        success = upload_config(server_addr, namespace, group, data_id, content)
        
        if success:
            print(f"✓ {data_id} uploaded successfully")
        else:
            print(f"✗ {data_id} upload failed")
            return False
    
    return True

if __name__ == "__main__":
    import argparse
    
    parser = argparse.ArgumentParser(description='Deploy Nacos configurations')
    parser.add_argument('--env', type=str, default='dev', 
                       choices=['dev', 'test', 'uat', 'prod'],
                       help='Target environment')
    
    args = parser.parse_args()
    
    success = deploy_configs(args.env)
    sys.exit(0 if success else 1)
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
基础设施集成测试
Requirements: Python 3.12+
"""
import sys
import requests
import pytest

# 检查Python版本
if sys.version_info < (3, 12):
    raise RuntimeError("Python 3.12+ is required")

BASE_URL = "http://localhost:18102"

class TestInfrastructure:
    """基础设施测试类"""
    
    def test_health_check(self):
        """测试健康检查接口"""
        response = requests.get(f"{BASE_URL}/actuator/health", timeout=10)
        assert response.status_code == 200
        data = response.json()
        assert data["status"] == "UP"
        
        # 验证各组件状态
        components = data.get("components", {})
        assert components.get("db", {}).get("status") == "UP"
        assert components.get("redis", {}).get("status") == "UP"
    
    def test_prometheus_metrics(self):
        """测试Prometheus监控指标"""
        response = requests.get(f"{BASE_URL}/actuator/prometheus", timeout=10)
        assert response.status_code == 200
        
        content = response.text
        # 验证关键指标存在
        assert "jvm_memory_used_bytes" in content
        assert "http_server_requests_seconds" in content
        assert "process_cpu_usage" in content
    
    def test_nacos_registration(self):
        """测试服务注册到Nacos"""
        # 查询Nacos服务列表
        nacos_url = "http://localhost:8848/nacos/v1/ns/instance/list"
        params = {
            "serviceName": "crm-operation-tenant",
            "namespaceId": "operation-dev",
            "groupName": "TENANT_GROUP"
        }
        
        response = requests.get(nacos_url, params=params, timeout=10)
        assert response.status_code == 200
        
        data = response.json()
        assert len(data.get("hosts", [])) > 0, "Service not registered in Nacos"
```

### 5.2 详细设计

**必需内容：**

#### 5.2.1 数据库设计

**表设计要求：**

1. **命名规范**
   - 表名：小写字母+下划线，如 `tenant_info`
   - 字段名：小写字母+下划线，如 `tenant_name`
   - 遵循[代码规范](./代码规范.md)中的数据库规范

2. **必需字段**
   ```sql
   id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '主键ID',
   tenant_id BIGINT NOT NULL COMMENT '租户ID',
   created_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
   updated_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
   created_by BIGINT COMMENT '创建人ID',
   updated_by BIGINT COMMENT '更新人ID',
   deleted TINYINT NOT NULL DEFAULT 0 COMMENT '逻辑删除标记 0-未删除 1-已删除',
   version INT NOT NULL DEFAULT 0 COMMENT '乐观锁版本号'
   ```

3. **索引设计**
   - 主键索引
   - 租户ID索引（必需）
   - 业务查询字段索引
   - 联合索引（针对多条件查询）

4. **数据隔离策略**
   
   **表隔离（租户敏感数据）：**
   - 线索数据：`t_lead_{tenant_id}`
   - 客户数据：`t_customer_{tenant_id}`
   - 员工数据：`t_employee_{tenant_id}`
   
   **行隔离（公共数据或配置数据）：**
   - 元数据：`t_lead_field_meta`（包含 `tenant_id` 字段）
   - 配置数据：`t_role_permission`（包含 `tenant_id` 字段）
   - 字典数据：`t_dict_data`（包含 `tenant_id` 字段）

```java
class TenantServiceTest {
    
    @Autowired
    private TenantService tenantService;
    
    @MockBean
    private TenantRepository tenantRepository;
    
    @Test
    @DisplayName("创建租户-成功")
    void testCreateTenant_Success() {
        // Given
        TenantDTO dto = new TenantDTO();
        dto.setTenantCode("TENANT001");
        dto.setTenantName("测试企业");
        
        // When
        Result<TenantVO> result = tenantService.createTenant(dto);

        dto.setTenantCode("TENANT001");

            tenantService.createTenant(dto);
        });
    }
}
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
数据库客户端工具
Requirements: Python 3.12+
"""
import sys
import pymysql
from typing import List, Dict, Any, Optional

# 检查Python版本
if sys.version_info < (3, 12):
    raise RuntimeError("Python 3.12+ is required")

class DBClient:
    """MySQL数据库客户端"""
    
    def __init__(self, host: str, port: int, user: str, password: str, database: str):
        self.connection = pymysql.connect(
            host=host,
            port=port,
            user=user,
            password=password,
            database=database,
            charset='utf8mb4',
            cursorclass=pymysql.cursors.DictCursor
        )
    
    def execute(self, sql: str, params: Optional[tuple] = None) -> int:
        """执行SQL语句"""
        with self.connection.cursor() as cursor:
            result = cursor.execute(sql, params)
            self.connection.commit()
            return result
    
    def query_one(self, sql: str, params: Optional[tuple] = None) -> Optional[Dict]:
        """查询单条记录"""
        with self.connection.cursor() as cursor:
            cursor.execute(sql, params)
            return cursor.fetchone()
    
    def query_all(self, sql: str, params: Optional[tuple] = None) -> List[Dict]:
        """查询多条记录"""
        with self.connection.cursor() as cursor:
            cursor.execute(sql, params)
            return cursor.fetchall()
    
    def close(self):
        """关闭连接"""
        if self.connection:
            self.connection.close()
```

## 10. 常见问题

### 10.1 Nacos配置不生效

**原因：**
- Namespace或Group配置错误
- 配置文件格式错误
- 未启用配置刷新

**解决方案：**
- 检查 `application.yml` 中的Namespace和Group配置
- 验证Nacos配置文件YAML语法
- 在需要刷新的Bean上添加 `@RefreshScope` 注解

### 10.2 单元测试运行缓慢

**原因：**
- 启动了完整的Spring容器
- 有大量的外部依赖
- 测试数据准备耗时

**解决方案：**
- 使用 `@WebMvcTest`、`@MybatisPlusTest` 等切片测试
- 使用Mock替代真实依赖
- 优化测试数据准备逻辑

### 10.3 集成测试不稳定

**原因：**
- 测试用例之间有依赖
- 测试数据未清理
- 并发执行冲突

**解决方案：**
- 保证测试用例独立性
- 每个测试后清理数据
- 使用事务回滚或隔离的测试数据
